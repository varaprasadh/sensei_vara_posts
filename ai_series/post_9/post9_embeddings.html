05 / 08
// the classic example

Vector math = semantics.

Directions in vector space carry meaning. You can do algebra on them.

embed("king") − embed("man") + embed("woman") ≈ embed("queen") 
// cosine similarity: how close are two vectors?
function cosine(a, b) {
  let dot = 0, na = 0, nb = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    na  += a[i] ** 2;
    nb  += b[i] ** 2;
  }
  return dot / (Math.sqrt(na) * Math.sqrt(nb));
}
// → 1.0 = identical, 0 = unrelated, -1 = opposite

Words have direction. Direction has meaning.
