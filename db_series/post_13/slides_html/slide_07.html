Check dead tuple counts and last vacuum time for your tables:

-- find tables with most dead tuples
SELECT
  relname,
  n_dead_tup,
  n_live_tup,
  last_vacuum,
  last_autovacuum
FROM   pg_stat_user_tables
ORDER BY n_dead_tup DESC
LIMIT  5;
-- vacuum + update stats on a table
VACUUM ANALYZE users;

-- check xid age (wraparound risk)
SELECT datname, age(datfrozenxid) AS xid_age
FROM   pg_database
ORDER BY xid_age DESC;
