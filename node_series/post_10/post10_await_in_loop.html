$ # the fix — Promise.all + map

Launch all at once.
Wait together.

  urls.map(async url => {

  })

$ # bounded concurrency — p-limit

Max 8 at a time.
Safe and fast.

import pLimit from 'p-limit';

const limit = pLimit(8);            // max 8 concurrent

const results = await Promise.all(
  urls.map(url => limit(() => fetch(url)))
);

$ # see it yourself — node benchmark

Paste. Run. See it.

const delay = ms => new Promise(r => setTimeout(r, ms));
const urls = Array.from({ length: 10 }, (_, i) => i);

// Sequential
let t = Date.now();
for (const u of urls) await delay(50);
console.log('sequential: ' + (Date.now() - t) + 'ms');  // ~500ms

// Parallel
t = Date.now();
await Promise.all(urls.map(u => delay(50)));
console.log('parallel:   ' + (Date.now() - t) + 'ms');  // ~50ms

// p-limit: concurrency=3
const limit = c => {
  let a = 0; const q = [];
  return fn => new Promise(res => {
    const run = () => { a++; fn().then(v => { a--; res(v); q.shift()?.(); }); };
    a < c ? run() : q.push(run);
  });
};
const lim = limit(3);
t = Date.now();
await Promise.all(urls.map(u => lim(() => delay(50))));
console.log('limited(3): ' + (Date.now() - t) + 'ms');  // ~200ms
sequential: ~500ms
parallel: ~50ms
limited(3): ~200ms
